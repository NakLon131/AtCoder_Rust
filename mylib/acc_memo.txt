let n = 5;
let val = vec![1, 2, 3, 4, 5];

// 前から累積和
// prefix_sum
let mut prefix_sum = vec![0; n+2];
for i in 0..n {
	prefix_sum[i+1] = prefix_sum[i];
	prefix_sum[i+1] += val[i];
}
dbg!(&prefix_sum);
// [0, 1, 3, 6, 10, 15]

// 後ろから累積和
// suffix_sum
let mut suffix_sum = vec![0; n+2];
for i in (0..n).rev() {
	suffix_sum[i+1] = suffix_sum[i+2];
	suffix_sum[i+1] += val[i];
}
dbg!(&suffix_sum);
// [0, 15, 14, 12, 9, 5, 0]
