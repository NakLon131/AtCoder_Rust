{
	// ベクトル 
	fn vectol(a: (f64, f64), b: (f64, f64)) -> (f64, f64) {
		(b.0-a.0, b.1-a.1)
	}
	// 線分
	fn len_size(a: (f64, f64), b: (f64, f64)) -> f64 {
		((b.0-a.0)*(b.0-a.0) + (b.1-a.1)*(b.1-a.1)).sqrt()
	}
	// ベクトルの内積
	fn inner(a: (f64, f64), b: (f64, f64)) -> f64 {
		a.0 * b.0 + a.1 * b.1
	}
	// ベクトルの外積
	fn outer(a: (f64, f64), b: (f64, f64)) -> f64 {
		a.0 * b.1 - a.1 * b.0
	}
}
