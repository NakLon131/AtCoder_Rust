// ACL E 
use ac_library::MinCostFlowGraph;

fn main() {
    input!{
		n: usize, k: i64,
		a: [[i64; n]; n],
	}
	// s, 縦番号(1^N), 横番号(N+1 ~ 2N), t
	let mut mf = MinCostFlowGraph::<i64>::new(2*n+2);
	mf.add_edge(0, 2*n+1, INF, INF);

	// sと縦のマッチング
	for i in 0..n {
		mf.add_edge(0, i+1, k, 0);
	}
	// 横とtのマッチング
	for i in 0..n {
		mf.add_edge(n+i+1, 2*n+1, k, 0);
	}

	let mut edges = vec![];
	// 縦と横のマッチング(辺も記録)
	for i in 0..n {
		for j in 0..n {
			edges.push(mf.add_edge(i+1, n+j+1, 1, INF - a[i][j]));
		}
	}

	let res = mf.flow(0, 2*n+1, INF);
	println!("{}", INF * res.0 - res.1);
	
	let mut state = vec![vec!['.'; n]; n];
	for e in edges {
		let e = mf.get_edge(e);
		if e.flow > 0 {
			state[e.from-1][e.to-1 - n] = 'X';
		}
	}
	println!("{}", state.into_iter().map(|s| s.iter().collect::<String>()).join("\n"));

}
